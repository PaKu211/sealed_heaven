constants = Path('src/tests/golden.rs').read_text()
document = source.decode()
published_vectors = re.findall(r'`([a-f0-9]{16,}|0x[a-f0-9]{8})`', document)
vector_set_from_code = set(re.findall(r'"([a-f0-9]{16,})"|\b(0x[a-f0-9]{8})\b', constants))

            set(document.replace('```text\n','').splitlines()))

code_checkpoints = re.findall(r'bytes: concat!\(\s*((?:"[a-f0-9]+",\s*)+)\),\s*sha256:\s*"([a-f0-9]+)"',constants)

    assert len(expected_bytes) == int(size)
